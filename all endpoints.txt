custom onetime urls endpoints creators:

Admin POST to: https://login.support-noreply.help/admin/otp/create
Admin POST to: https://login.support-noreply.help/admin/otp/update
Admin POST to: https://login.support-noreply.help/admin/otp/revoke
Admin GET to:  https://login.support-noreply.help/admin/otp/status


# 1. Create OTP
curl -X POST https://login.support-noreply.help/admin/otp/create \
  -H "X-Admin-Token: $ADMIN_TOKEN" \
  -d '{
    "clientId": "client-2",
    "expiresIn": 500,
    "maxUses": 1,
    "redirectPath": "/testing",
    "customId": "testing"
  }'

# 2. Check status
curl -H "X-Admin-Token: $ADMIN_TOKEN" \
  "https://login.support-noreply.help/admin/otp/status?custom_id=testing"

# 3. Reset uses to 5
curl -X POST https://login.support-noreply.help/admin/otp/update \
  -H "X-Admin-Token: $ADMIN_TOKEN" \
  -d '{"custom_id": "testing", "action": "reset_uses", "value": 1}'

# 4. Extend expiry by 1 hour (3600 seconds)
curl -X POST https://login.support-noreply.help/admin/otp/update \
  -H "X-Admin-Token: $ADMIN_TOKEN" \
  -d '{"custom_id": "test", "action": "extend", "value": 500}'

# 5. Change redirect path
curl -X POST https://login.support-noreply.help/admin/otp/update \
  -H "X-Admin-Token: $ADMIN_TOKEN" \
  -d '{"custom_id": "bakchod_1", "action": "change_path", "value": "/new-path"}'

# 6. Revoke token
curl -X POST https://login.support-noreply.help/admin/otp/revoke \
  -H "X-Admin-Token: $ADMIN_TOKEN" \
  -d '{"custom_id": "bakchod_1"}'
